<p class="club-lead-intro">The <strong>2025 season</strong> was <a href="/clubs/locomotive/">Locomotive</a>'s third season in their history.</p>

Locomotive played in five competitions thanks to the successful 2024 campaign which saw the club win the league title and the Coupe du Québec.
This season is the first under new head coach Danny Wilson, who was appointed on December 30 2024, following Ilan Mercier's departure to MLS side D.C. United at the end of the 2024 season.

In February 2025, Locomotive played for the first time in the CONCACAF Champions Cup. The first round was played against CS Herediano, Costa Rican side, and being the first La Ligue club to win a round in the competition. Locomotive fell in the Second Round versus Seattle Sounders FC with a 3-2 loss in aggregate.

On April 5 2025, Locomotive were presented La Supercoupe after a 2-0 win against Métabéroutin, the 2024 La Ligue second placed team.

In October 2025, Locomotive were crowned La Ligue champions for a second season in a row. Danny Wilson became the second manager in history to win La Ligue in their first season.

      var cell = row.children[index];

    }

    function toComparable(text, isNumeric) {
      if (isNumeric) {
        var num = Number(text.replace(/[^\d.-]/g, ""));
        return Number.isFinite(num) ? num : -Infinity;
      }

    }

    function sortWithinGroups(columnIndex, direction) {
      var rows = Array.from(tbody.querySelectorAll("tr"));

      var groups = [];
      var current = null;

      rows.forEach(function (row) {
        if (row.classList.contains("squad-group-row")) {
          current = { header: row, items: [] };
          groups.push(current);
        } else if (current) {
          current.items.push(row);
        }
      });

      if (groups.length === 0) return;

      var isNumeric = columnIndex === 0 || columnIndex === 6;

      groups.forEach(function (group) {
        group.items.sort(function (a, b) {
          var aValue = toComparable(getCellText(a, columnIndex), isNumeric);
          var bValue = toComparable(getCellText(b, columnIndex), isNumeric);
          if (aValue < bValue) return direction === "asc" ? -1 : 1;
          if (aValue > bValue) return direction === "asc" ? 1 : -1;
          return 0;
        });
      });

      groups.forEach(function (group) {
        tbody.appendChild(group.header);
        group.items.forEach(function (row) {
          tbody.appendChild(row);
        });
      });
    }

    headers.forEach(function (header, index) {
      header.style.cursor = "pointer";
      header.title = "Sort";
